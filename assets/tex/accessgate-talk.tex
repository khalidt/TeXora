% Before ltx-talk 0.6.12 (2026-11-12) its own contents page breaks the tag structure (its toc
% tagging sockets), so there the contents are listed plainly: sections, and subsections
% indented. [currentsection] etc. are ignored. From 0.6.12 its own contents page is used.
\@ifclasslater{ltx-talk}{2026-11-12}{}{%
\DeclareDocumentCommand\tableofcontents{o}{\begingroup
  % Each entry is a plain paragraph: the kernel's TOCI around a paragraph is not allowed in PDF/UA-2.
  \def\contentsline##1##2##3##4{\csname l@##1\endcsname{##2}{##3}}%
  \def\numberline##1{##1\enspace}%
  \def\l@section##1##2{\par\noindent\textbf{##1}\par}%
  \def\l@subsection##1##2{\par\noindent\hspace*{2em}##1\par}%
  \def\l@subsubsection##1##2{}%
  \makeatletter\InputIfFileExists{\jobname.toc}{}{}\par
  \if@filesw\@ifundefined{acc@tocopen}{\gdef\acc@tocopen{}%
    \AddToHook{enddocument/afterlastpage}{\expandafter\newwrite\csname tf@toc\endcsname
      \immediate\openout\csname tf@toc\endcsname\jobname.toc\relax}}{}\fi
  \endgroup}}
% \appendix (written \AccAppendix in the copy): before 0.6.12 ltx-talk's broke the tag structure
% between frames, so there it does nothing.
\@ifclasslater{ltx-talk}{2026-11-12}{\let\AccAppendix\appendix}{\let\AccAppendix\relax}
% Frame options the copy uses: ltx-talk 0.6 knows label= (\againframe repeats the frame) and
% auto-break (for allowframebreaks); before, both are accepted and do nothing, and \againframe
% (above) repeats nothing.
\ExplSyntaxOn
\keys_if_exist:nnF { talk / frame } { label }
  { \keys_define:nn { talk / frame } { label .code:n = { } } }
\keys_if_exist:nnF { talk / frame } { auto-break }
  { \keys_define:nn { talk / frame } { auto-break .code:n = { } } }
\ExplSyntaxOff
% Beamer defines these theorem environments; a presentation's own \newtheorem comes first.
\RequirePackage{amsthm}
\AddToHook{begindocument/before}{%
  \@ifundefined{theorem}{\newtheorem{theorem}{Theorem}}{}%
  \@ifundefined{corollary}{\newtheorem{corollary}{Corollary}}{}%
  \@ifundefined{lemma}{\newtheorem{lemma}{Lemma}}{}%
  \@ifundefined{fact}{\newtheorem{fact}{Fact}}{}%
  \theoremstyle{definition}%
  \@ifundefined{definition}{\newtheorem{definition}{Definition}}{}%
  \@ifundefined{definitions}{\newtheorem{definitions}{Definitions}}{}%
  \@ifundefined{example}{\newtheorem{example}{Example}}{}%
  \@ifundefined{examples}{\newtheorem{examples}{Examples}}{}}
% With LuaLaTeX ltx-talk uses unicode-math, which has amssymb's symbols (amssymb itself clashes);
% two of them have other names there.
\AddToHook{begindocument/end}{%
  \@ifundefined{square}{\@ifundefined{mdlgwhtsquare}{}{\let\square\mdlgwhtsquare}}{}%
  \@ifundefined{blacksquare}{\@ifundefined{mdlgblksquare}{}{\let\blacksquare\mdlgblksquare}}{}}
\makeatother
